# WAMR built from the pinned source tarball. Kept in a directory of its own because WAMR's build scripts set
# include directories and definitions for everything in the directory they run in.

basalt_wamr_source(WAMR_ROOT_DIR)

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _processor)
if(_processor MATCHES "^(x86_64|amd64)$")
    set(WAMR_BUILD_TARGET X86_64)
elseif(_processor MATCHES "^(aarch64|arm64)$")
    set(WAMR_BUILD_TARGET AARCH64)
else()
    message(FATAL_ERROR "WAMR backend: unsupported processor ${CMAKE_SYSTEM_PROCESSOR}")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    set(WAMR_BUILD_PLATFORM freebsd)
elseif(APPLE)
    set(WAMR_BUILD_PLATFORM darwin)
elseif(ANDROID)
    set(WAMR_BUILD_PLATFORM android)
elseif(WIN32)
    set(WAMR_BUILD_PLATFORM windows)
else()
    set(WAMR_BUILD_PLATFORM linux)
endif()

set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 1)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_TAIL_CALL 1)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
if(NOT DEFINED BASALT_WAMR_THREAD_MGR)
    set(BASALT_WAMR_THREAD_MGR 1)
endif()
set(WAMR_BUILD_THREAD_MGR ${BASALT_WAMR_THREAD_MGR})

set(BASALT_WAMR_FAST_JIT OFF)
set(BASALT_WAMR_GC OFF)
if(BASALT_WAMR_ENABLE_GC)
    set(BASALT_WAMR_GC ON)
elseif(BASALT_WAMR_FAST_JIT_MODE STREQUAL "ON" OR
       (BASALT_WAMR_FAST_JIT_MODE STREQUAL "AUTO" AND WAMR_BUILD_TARGET STREQUAL "X86_64" AND NOT WIN32))
    set(BASALT_WAMR_FAST_JIT ON)
endif()

if(BASALT_WAMR_GC)
    set(WAMR_BUILD_GC 1)
else()
    set(WAMR_BUILD_GC 0)
endif()
if(BASALT_WAMR_FAST_JIT)
    set(WAMR_BUILD_FAST_JIT 1)
    set(WAMR_BUILD_FAST_INTERP 0)
else()
    set(WAMR_BUILD_FAST_JIT 0)
    set(WAMR_BUILD_FAST_INTERP 1)
endif()

if(BASALT_WAMR_FAST_JIT)
    basalt_wamr_asmjit_source(_asmjit_dir)
    set(FETCHCONTENT_SOURCE_DIR_ASMJIT "${_asmjit_dir}" CACHE PATH "" FORCE)
    set(FETCHCONTENT_FULLY_DISCONNECTED ON)
endif()

include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)

add_library(basalt_wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(basalt_wamr SYSTEM PUBLIC ${WAMR_ROOT_DIR}/core/iwasm/include)
target_compile_options(basalt_wamr PRIVATE -w)
set_target_properties(basalt_wamr PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(basalt_wamr PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
if(BASALT_WAMR_FAST_JIT)
    target_compile_definitions(basalt_wamr INTERFACE BASALT_WAMR_FAST_JIT)
endif()
add_library(basalt::wamr ALIAS basalt_wamr)
